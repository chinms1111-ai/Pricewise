CORS(app)


@app.route("/")
def home():
    return render_template("index.html")

 

def get_db():
    # opens the door to the database and allows us to interact with it
    conn = sqlite3.connect('pricewise.db')
    
    # it makes the database return as a neat dictionary instead of a messy list, so we can easily access the data by column names
    conn.row_factory = sqlite3.Row
    
    # hands the connection back to whoever called it
    return conn


# when someone knocks/add products and sends data flask answer
# POST means they are sending info not just asking for it
@app.route("/add_product", methods=["POST"])

def add_product():
    data = request.json
    if not data:
        return jsonify({"error": "No data received"}), 400
    
    name = data.get("name","").strip()
    
    if not name:
        return jsonify({"error": "Product name is required"}), 400
    
    conn = get_db()
    c = conn.cursor()
    c.execute("INSERT INTO products (name, url) VALUES (?, ?)",
              (name , data.get("url", "")))
    
    conn.commit()
    product_id = c.lastrowid
    conn.close()
    return jsonify({"message": "Product added", "id ": product_id})


 

@app.route("/log_price",methods = ["POST"])
def log_price():
    data = request.json
    conn = get_db()
    c = conn.cursor()
    c.execute("INSERT INTO price history (product_id ,price ,date) VALUES (?,?,?)",
              (data["product_id"] , data["price"] , str(date.today())))
    conn.commit()
    conn.close()
    return jsonify({"message":"Price logged"})


# this time we are not sending data using POST we are now asking for data using GET
@app.route("/products" , methods = ["GET"])

def get_products():
    conn = get_db()
    c = conn.cursor()
    
    # telling the pen to get everything from the product table 
    c.execute("SELECT * FROM products")
    
    # ot converts every row into a clean dictionary like gets all the row and converts them to another
    products = [dict(row) for row in c.fetchall()]
    
    
    conn.close()
    return jsonify(products)

# the <int:product_id> means the url carries a number like history 1 or history 5 from another time
# and product_id help its only ge details from specified products not everything at once
@app.route("/history/<int:product_id>", methods= ["GET"])
def get_history(product_id):
    conn = get_db()
    c = conn.cursor()
    
    # order by date ASC sort the oldest to newest so the graph shows the direction
    c.execute("SELECT * FROM price_history WHERE  product_id = ? ORDER BY date ASC",
                 (product_id))
    
    history = [dict(row) for row in c.fetchall()]
    conn.close()
    return jsonify(history)
